   160→   160→  async function applyCoupon() {

   169→   169→        method: "POST",
   170→   170→        headers: { "Content-Type": "application/json" },
   171→   171→        body: JSON.stringify({ code: couponCode.trim(), subtotal }),
   172→   172→      });
   173→   173→      const data = await res.json();
   174→   174→      if (data.success) {
   175→   175→        // Merge coupon object + discount + freeShipping into one state object
   176→   176→        setCouponData({
   177→   177→          ...data.coupon,
   178→   178→          discount: data.discount,
   179→   179→          freeShipping: data.freeShipping,
   180→   180→        });
   181→   181→        toast.success(
   182→   182→          locale === "bn"
   183→   183→            ? `কুপন প্রয়োগ হয়েছে! আপনি ${formatPrice(data.discount)} সাশ্রয় করেছেন`
   184→   184→            : `Coupon applied! You saved ${formatPrice(data.discount)}`
   185→   185→        );
   186→   186→      } else {
   187→   187→        toast.error(data.error || "Invalid coupon code");
   188→   188→      }
   189→   189→    } catch {
   190→   190→      toast.error("Failed to validate coupon");
   191→   191→    } finally {
   192→   192→      setCouponLoading(false);
   193→   193→    }
   194→   194→  }
