   107→-- Auto-create profile when a new auth user signs up
   108→CREATE OR REPLACE FUNCTION public.handle_new_user()
   109→RETURNS TRIGGER AS $$
   110→BEGIN
   111→  INSERT INTO public.profiles (id, email, name, image)
   112→  VALUES (
   113→    NEW.id,
   114→    NEW.email,
   115→    COALESCE(NEW.raw_user_meta_data->>'full_name', NEW.raw_user_meta_data->>'name', split_part(NEW.email, '@', 1)),
   116→    NEW.raw_user_meta_data->>'avatar_url'
   117→  )
   118→  ON CONFLICT (id) DO NOTHING;
   119→  RETURN NEW;
   120→END;
   121→$$ LANGUAGE plpgsql SECURITY DEFINER SET search_path = public;

   124→CREATE TRIGGER on_auth_user_created

   459→-- Auto-generate order number
   460→CREATE OR REPLACE FUNCTION generate_order_number()
   461→RETURNS TRIGGER AS $$
   462→BEGIN
   463→  IF NEW.order_number IS NULL THEN
   464→    NEW.order_number := 'RPH-' || TO_CHAR(now(), 'YYMMDD') || '-' || LPAD((EXTRACT(EPOCH FROM now())::bigint % 100000)::text, 5, '0');
   465→  END IF;
   466→  RETURN NEW;
   467→END;
   468→$$ LANGUAGE plpgsql;

   470→DROP TRIGGER IF EXISTS set_order_number ON orders;
   471→CREATE TRIGGER set_order_number
   472→  BEFORE INSERT ON orders
   473→  FOR EACH ROW EXECUTE PROCEDURE generate_order_number();

   743→-- ============================================
   744→-- UPDATED_AT TRIGGER (auto-update on row change)
   745→-- ============================================
   746→
   747→CREATE OR REPLACE FUNCTION update_updated_at_column()
   748→RETURNS TRIGGER AS $$
   749→BEGIN
   750→  NEW.updated_at = now();
   751→  RETURN NEW;
   752→END;
   753→$$ LANGUAGE plpgsql;
   754→
   755→DO $$
   756→DECLARE
   757→  tbl TEXT;
   758→BEGIN
   759→  FOR tbl IN
   760→    SELECT unnest(ARRAY[
   761→      'profiles', 'categories', 'brands', 'products', 'product_variants',
   762→      'bundles', 'carts', 'addresses', 'orders', 'coupons', 'reviews',
   763→      'banners', 'blog_posts', 'support_tickets', 'settings'
   764→    ])
   765→  LOOP
   766→    EXECUTE format('
   767→      DROP TRIGGER IF EXISTS set_updated_at ON %I;
   768→      CREATE TRIGGER set_updated_at BEFORE UPDATE ON %I
   769→      FOR EACH ROW EXECUTE PROCEDURE update_updated_at_column();
   770→    ', tbl, tbl);
   771→  END LOOP;
   772→END $$;
