   153→      // If product has a real ID (not empty), it's an edit (PUT)
   154→      // Otherwise it's a new product (POST)

   156→      const method = isEditing ? "PUT" : "POST";

   162→        method,
   163→        headers: { "Content-Type": "application/json" },
   164→        body: JSON.stringify({
   165→          ...product,
   166→          brandId: null,
   167→          categoryId: product.category_id,
   168→          shortDescription: product.short_description,
   169→          // Explicitly map discount_price (DB snake_case) → discountPrice (API camelCase)
   170→          discountPrice: product.discount_price ?? null,
   171→          costPrice: (product as any).cost_price ?? null,
   172→          isFeatured: product.is_featured,
   173→          isBestSeller: product.is_best_seller,
   174→          isNewArrival: product.is_new_arrival,
   175→          isFlashSale: product.is_flash_sale,

   178→            url: img.url,
   179→            isPrimary: img.is_primary,
   180→            altText: product.name,

   183→          sizes: (product.sizes || []).map((s: any) => ({
   184→            size: s.size,
   185→            stock: s.stock,
   186→            sortOrder: s.sort_order ?? 0,

   196→          searchKeywords: product.search_keywords || null,

   214→      setSaving(false);
   215→    }

   462→/**
   463→ * Product Form Dialog — full form with sizes, colors, stock management
   464→ */
   465→function ProductFormDialog({
   466→  product,
   467→  categories,
   468→  open,
   469→  onOpenChange,
   470→  onSave,
   471→  saving,
   472→  isEditing,
   473→}: {
   474→  product: Product;
   475→  categories: Category[];
   476→  open: boolean;
   477→  onOpenChange: (open: boolean) => void;
   478→  onSave: (product: Product) => void;
   479→  saving: boolean;
   480→  isEditing: boolean;
   481→}) {
   482→  const [form, setForm] = React.useState<Product>(product);
   483→
   484→  React.useEffect(() => {
   485→    setForm(product);
   486→  }, [product]);
   487→
   488→  function update<K extends keyof Product>(key: K, value: Product[K]) {
   489→    setForm((prev) => ({ ...prev, [key]: value }));
   490→  }
   491→
   492→  // Size management
   493→  function addSize(size: string) {
   494→    if (!size) return;
   495→    if (form.sizes?.some((s) => s.size === size)) {
   496→      toast.error("Size already added");
   497→      return;
   498→    }
   499→    update("sizes", [...(form.sizes || []), { size, stock: 0 }]);
   500→  }
   501→
   502→  function updateSizeStock(idx: number, stock: number) {
   503→    const sizes = [...(form.sizes || [])];
   504→    sizes[idx] = { ...sizes[idx], stock };
   505→    update("sizes", sizes);
   506→  }
   507→
   508→  function removeSize(idx: number) {
   509→    const sizes = (form.sizes || []).filter((_, i) => i !== idx);
   510→    update("sizes", sizes);
   511→  }
   512→
   513→  // Color management
   514→  function addColor(name: string, hex: string) {
   515→    if (!name || !hex) return;
   516→    if (form.colors?.some((c) => c.name === name)) {
   517→      toast.error("Color already added");
   518→      return;
   519→    }
   520→    update("colors", [...(form.colors || []), { name, hex_value: hex, stock: 0 }]);
   521→  }
   522→
   523→  function updateColorStock(idx: number, stock: number) {
   524→    const colors = [...(form.colors || [])];
   525→    colors[idx] = { ...colors[idx], stock };
   526→    update("colors", colors);
   527→  }
   528→
   529→  function removeColor(idx: number) {
   530→    const colors = (form.colors || []).filter((_, i) => i !== idx);
   531→    update("colors", colors);
   532→  }
   533→
   534→  function handleSubmit(e: React.FormEvent) {
   535→    e.preventDefault();
   536→    onSave(form);
   537→  }
   538→
   539→  return (
   540→    <Dialog open={open} onOpenChange={onOpenChange}>
   541→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   542→        <DialogHeader>
   543→          <DialogTitle>{isEditing ? "Edit Product" : "Add New Product"}</DialogTitle>
   544→        </DialogHeader>
   545→
   546→        <form id="product-form" onSubmit={handleSubmit} className="space-y-4">
   547→          {/* Basic Info */}
   548→          <div className="space-y-3">
   549→            <h3 className="text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   550→              Basic Information
   551→            </h3>
   552→            <div className="space-y-2">
   553→              <Label htmlFor="name">Product Name *</Label>
   554→              <Input
   555→                id="name"
   556→                required
   557→                value={form.name}
   558→                onChange={(e) => update("name", e.target.value)}
   559→              />
   560→            </div>
   561→            <div className="grid grid-cols-2 gap-3">
   562→              <div className="space-y-2">
   563→                <Label htmlFor="sku">SKU *</Label>
   564→                <Input
   565→                  id="sku"
   566→                  required
   567→                  value={form.sku}
   568→                  onChange={(e) => update("sku", e.target.value)}
   569→                />
   570→              </div>
   571→              <div className="space-y-2">
   572→                <Label htmlFor="slug">Slug (URL)</Label>
   573→                <Input
   574→                  id="slug"
   575→                  placeholder="auto-generated if empty"
   576→                  value={form.slug}
   577→                  onChange={(e) => update("slug", e.target.value)}
   578→                />
